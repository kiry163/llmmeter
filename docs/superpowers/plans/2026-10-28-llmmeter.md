# llmmeter Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 一个本机运行的 OpenAI 兼容透传网关，记录每次调用的 token 与按峰谷窗口计价的花费，并提供只读统计页面。

**Architecture:** 单进程 Go：`net/http` 手写反向代理，只在必要路径上注入 `stream_options`，响应体经 tee 旁路解析 SSE/JSON，usage 归一化后按“base 价 + 时间窗覆盖”计价，写入纯 Go SQLite；embed 页面用 SVG 画折线。

**Tech Stack:** Go 1.24、`gopkg.in/yaml.v3`、`modernc.org/sqlite`、`embed`、标准库 `net/http/httptest`。

**Spec:** `docs/superpowers/specs/2026-10-28-llm-meter-design.md`

## Global Constraints

- 模块路径 `github.com/kiry163/llmmeter`；二进制名 `llmmeter`。
- 只支持 `/v1/chat/completions` 与 `/v1/responses` 的 usage 解析；其它路径字节级直通且不落库。
- 成本用整数纳美元：`nano = round(tokens × rate_per_mtok × 1000)`，累加后再换算 USD。
- 时间窗按请求 `completed_at` 换算到配置时区判定；存储 UTC，另存配置时区的 `local_day` 供按自然日聚合。
- 拿不到 usage 时不估算：`usage_status` 记 `no_usage|parse_failed|too_large`。
- 中文注释可以，标识符一律英文；错误信息英文。
- 不引入前端构建链与第三方前端资产（与 spec 的差异：手写 SVG 折线替代 Chart.js）。

## Review Focus

1. SSE 行被 TCP 分片切断、`data:` 与空行混排、上游发心跳注释行 → 必须不丢 usage、不阻塞转发。
2. 客户端强制 `Accept-Encoding: gzip` 且上游仍压缩 → 解析失败时按 `parse_failed` 记，进程不崩。
3. 请求体非 JSON、`stream` 缺失、或已显式设置 `stream_options` → 不得改写字节、不得 400。
4. 跨午夜窗口/周末边界（`Asia/Shanghai`）→ 命中窗口与成本正确。
5. 客户端提前断开而上游继续计费 → 行必须落库，尽量带 usage。

// internal/config/config.go
package config

import (
	"errors"
	"fmt"
	"os"
	"regexp"
	"strings"
	"time"

	"gopkg.in/yaml.v3"
)

type Duration time.Duration

func (d *Duration) Std() time.Duration { return time.Duration(*d) }

func (d *Duration) UnmarshalYAML(n *yaml.Node) error {
	var s string
	if err := n.Decode(&s); err != nil {
		return err
	}
	v, err := time.ParseDuration(s)
	if err != nil {
		return fmt.Errorf("duration %q: %w", s, err)
	}
	*d = Duration(v)
	return nil
}

type Config struct {
	Listen     string   `yaml:"listen"`
	GatewayKey string   `yaml:"gateway_key"`
	Timezone   string   `yaml:"timezone"`
	Upstream   Upstream `yaml:"upstream"`
	Pricing    Pricing  `yaml:"pricing"`
}

type Upstream struct {
	BaseURL string   `yaml:"base_url"`
	APIKey  string   `yaml:"api_key"`
	Timeout Duration `yaml:"timeout"`
}

type Pricing struct {
	Models map[string]Model `yaml:"models"`
}

type Model struct {
	InputPerMTok      float64  `yaml:"input_per_mtok"`
	OutputPerMTok     float64  `yaml:"output_per_mtok"`
	CacheReadPerMTok  float64  `yaml:"cache_read_per_mtok"`
	CacheWritePerMTok float64  `yaml:"cache_write_per_mtok"`
	ReasoningPerMTok  float64  `yaml:"reasoning_per_mtok"`
	Windows           []Window `yaml:"windows"`
}

type Window struct {
	Label             string   `yaml:"label"`
	Days              []string `yaml:"days"`
	Start             string   `yaml:"start"`
	End               string   `yaml:"end"`
	InputPerMTok      *float64 `yaml:"input_per_mtok"`
	OutputPerMTok     *float64 `yaml:"output_per_mtok"`
	CacheReadPerMTok  *float64 `yaml:"cache_read_per_mtok"`
	CacheWritePerMTok *float64 `yaml:"cache_write_per_mtok"`
	ReasoningPerMTok  *float64 `yaml:"reasoning_per_mtok"`
}

var envPattern = regexp.MustCompile(`\$\{env:([A-Za-z_][A-Za-z0-9_]*)\}`)

func expandEnv(s string) (string, error) {
	var missing []string
	out := envPattern.ReplaceAllStringFunc(s, func(m string) string {
		name := envPattern.FindStringSubmatch(m)[1]
		v, ok := os.LookupEnv(name)
		if !ok {
			missing = append(missing, name)
			return ""
		}
		return v
	})
	if len(missing) > 0 {
		return "", fmt.Errorf("missing environment variables: %s", strings.Join(missing, ", "))
	}
	return out, nil
}

func Load(path string) (*Config, error) {
	raw, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	text, err := expandEnv(string(raw))
	if err != nil {
		return nil, err
	}
	cfg := &Config{
		Listen:   "127.0.0.1:8100",
		Timezone: "Asia/Shanghai",
	}
	if err := yaml.Unmarshal([]byte(text), cfg); err != nil {
		return nil, fmt.Errorf("parse config: %w", err)
	}
	if cfg.GatewayKey == "" {
		return nil, errors.New("gateway_key is required")
	}
	if cfg.Upstream.BaseURL == "" || cfg.Upstream.APIKey == "" {
		return nil, errors.New("upstream.base_url and upstream.api_key are required")
	}
	if cfg.Upstream.Timeout == 0 {
		cfg.Upstream.Timeout = Duration(300 * time.Second)
	}
	if _, err := time.LoadLocation(cfg.Timezone); err != nil {
		return nil, fmt.Errorf("timezone %q: %w", cfg.Timezone, err)
	}
	for name, m := range cfg.Pricing.Models {
		for i, w := range m.Windows {
			if _, _, err := ParseClock(w.Start); err != nil {
				return nil, fmt.Errorf("model %s window %d start: %w", name, i, err)
			}
			if _, _, err := ParseClock(w.End); err != nil {
				return nil, fmt.Errorf("model %s window %d end: %w", name, i, err)
			}
			if w.Label == "" {
				return nil, fmt.Errorf("model %s window %d: label is required", name, i)
			}
			for _, d := range w.Days {
				if _, ok := Weekday(d); !ok {
					return nil, fmt.Errorf("model %s window %d: bad day %q", name, i, d)
				}
			}
		}
	}
	return cfg, nil
}

// ParseClock parses "HH:MM" (24h) into hours and minutes.
func ParseClock(s string) (h, m int, err error) {
	if len(s) != 5 || s[2] != ':' {
		return 0, 0, fmt.Errorf("clock %q must be HH:MM", s)
	}
	if _, err := fmt.Sscanf(s, "%02d:%02d", &h, &m); err != nil {
		return 0, 0, fmt.Errorf("clock %q: %w", s, err)
	}
	if h < 0 || h > 24 || m < 0 || m > 59 || (h == 24 && m != 0) {
		return 0, 0, fmt.Errorf("clock %q out of range", s)
	}
	return h, m, nil
}

var weekdays = map[string]time.Weekday{
	"sun": time.Sunday, "mon": time.Monday, "tue": time.Tuesday, "wed": time.Wednesday,
	"thu": time.Thursday, "fri": time.Friday, "sat": time.Saturday,
}

func Weekday(s string) (time.Weekday, bool) {
	d, ok := weekdays[strings.ToLower(s)]
	return d, ok
}

// internal/pricing/pricing.go
package pricing

import (
	"fmt"
	"math"
	"strings"
	"time"

	"github.com/kiry163/llmmeter/internal/config"
)

type Rates struct {
	Input      float64
	Output     float64
	CacheRead  float64
	CacheWrite float64
	Reasoning  float64
}

type partial struct {
	input, output, cacheRead, cacheWrite, reasoning *float64
}

type window struct {
	label     string
	days      uint8 // bit i = time.Weekday(i); 0 means every day
	startMin  int
	endMin    int
	overrides partial
}

type compiledModel struct {
	base    Rates
	windows []window
}

type Catalog struct {
	models map[string]compiledModel
}

func New(p config.Pricing) (*Catalog, error) {
	c := &Catalog{models: make(map[string]compiledModel, len(p.Models))}
	for name, m := range p.Models {
		cm := compiledModel{base: Rates{
			Input: m.InputPerMTok, Output: m.OutputPerMTok,
			CacheRead: m.CacheReadPerMTok, CacheWrite: m.CacheWritePerMTok,
			Reasoning: m.ReasoningPerMTok,
		}}
		for i, w := range m.Windows {
			sh, sm, err := config.ParseClock(w.Start)
			if err != nil {
				return nil, fmt.Errorf("%s window %d: %w", name, i, err)
			}
			eh, em, err := config.ParseClock(w.End)
			if err != nil {
				return nil, fmt.Errorf("%s window %d: %w", name, i, err)
			}
			var mask uint8
			if len(w.Days) == 0 {
				mask = 0
			} else {
				for _, d := range w.Days {
					wd, ok := config.Weekday(d)
					if !ok {
						return nil, fmt.Errorf("%s window %d: bad day %q", name, i, d)
					}
					mask |= 1 << uint(wd)
				}
			}
			cm.windows = append(cm.windows, window{
				label: w.Label, days: mask,
				startMin: sh*60 + sm, endMin: eh*60 + em,
				overrides: partial{
					input: w.InputPerMTok, output: w.OutputPerMTok,
					cacheRead: w.CacheReadPerMTok, cacheWrite: w.CacheWritePerMTok,
					reasoning: w.ReasoningPerMTok,
				},
			})
		}
		c.models[name] = cm
	}
	return c, nil
}

// Price returns the rates in effect at at, the matching window label ("base"
// when none matches), and whether the model is priced.
func (c *Catalog) Price(model string, at time.Time, loc *time.Location) (Rates, string, bool) {
	cm, ok := c.models[model]
	if !ok {
		return Rates{}, "", false
	}
	local := at.In(loc)
	for _, w := range cm.windows {
		if w.match(local) {
			return w.apply(cm.base), w.label, true
		}
	}
	return cm.base, "base", true
}

func (w window) match(t time.Time) bool {
	minute := t.Hour()*60 + t.Minute()
	dayBit := uint8(1) << uint(t.Weekday())
	allowed := func(d time.Weekday) bool {
		return w.days == 0 || w.days&(1<<uint(d)) != 0
	}
	if w.startMin < w.endMin {
		return allowed(t.Weekday()) && minute >= w.startMin && minute < w.endMin
	}
	// Wraps past midnight: same-day tail, or spill from the previous day.
	if allowed(t.Weekday()) && minute >= w.startMin {
		return true
	}
	prev := (t.Weekday() + 6) % 7
	return allowed(prev) && minute < w.endMin
}

func (w window) apply(base Rates) Rates {
	r := base
	if w.overrides.input != nil {
		r.Input = *w.overrides.input
	}
	if w.overrides.output != nil {
		r.Output = *w.overrides.output
	}
	if w.overrides.cacheRead != nil {
		r.CacheRead = *w.overrides.cacheRead
	}
	if w.overrides.cacheWrite != nil {
		r.CacheWrite = *w.overrides.cacheWrite
	}
	if w.overrides.reasoning != nil {
		r.Reasoning = *w.overrides.reasoning
	}
	return r
}

// FromYAMLName normalizes model names for lookup: exact match wins, otherwise
// a provider-prefixed name such as "deepseek/deepseek-chat" falls back to the
// suffix after the last slash.
func (c *Catalog) FromYAMLName(name string) (string, bool) {
	if _, ok := c.models[name]; ok {
		return name, true
	}
	if i := strings.LastIndexByte(name, '/'); i >= 0 {
		tail := name[i+1:]
		if _, ok := c.models[tail]; ok {
			return tail, true
		}
	}
	return "", false
}

// Cost returns the request cost in nano-USD.
func Cost(r Rates, prompt, completion, cacheRead, cacheWrite, reasoning int64) int64 {
	if cacheRead > prompt {
		cacheRead = prompt
	}
	if reasoning > completion {
		reasoning = completion
	}
	nano := func(tokens int64, rate float64) int64 {
		return int64(math.Round(float64(tokens) * rate * 1000))
	}
	total := nano(prompt-cacheRead, r.Input)
	total += nano(cacheRead, r.CacheRead)
	total += nano(cacheWrite, r.CacheWrite)
	if r.Reasoning > 0 {
		total += nano(reasoning, r.Reasoning)
		total += nano(completion-reasoning, r.Output)
	} else {
		total += nano(completion, r.Output)
	}
	return total
}

// internal/usage/usage.go
package usage

import "encoding/json"

type Usage struct {
	Prompt      int64 `json:"prompt_tokens"`
	Completion  int64 `json:"completion_tokens"`
	CacheRead   int64 `json:"cache_read_tokens"`
	CacheWrite  int64 `json:"cache_write_tokens"`
	Reasoning   int64 `json:"reasoning_tokens"`
	Total       int64 `json:"total_tokens"`
}

type chatUsage struct {
	PromptTokens        int64 `json:"prompt_tokens"`
	CompletionTokens    int64 `json:"completion_tokens"`
	TotalTokens         int64 `json:"total_tokens"`
	PromptCacheHit      int64 `json:"prompt_cache_hit_tokens"`
	PromptTokensDetails *struct {
		CachedTokens int64 `json:"cached_tokens"`
	} `json:"prompt_tokens_details"`
	CompletionTokensDetails *struct {
		ReasoningTokens int64 `json:"reasoning_tokens"`
	} `json:"completion_tokens_details"`
}

type chatEnvelope struct {
	Usage *chatUsage `json:"usage"`
}

type responsesEnvelope struct {
	Type     string `json:"type"`
	Response *struct {
		Usage *struct {
			InputTokens        int64 `json:"input_tokens"`
			OutputTokens       int64 `json:"output_tokens"`
			TotalTokens        int64 `json:"total_tokens"`
			InputTokensDetails *struct {
				CachedTokens int64 `json:"cached_tokens"`
			} `json:"input_tokens_details"`
			OutputTokensDetails *struct {
				ReasoningTokens int64 `json:"reasoning_tokens"`
			} `json:"output_tokens_details"`
		} `json:"usage"`
	} `json:"response"`
}

func (c chatUsage) normalize() Usage {
	u := Usage{
		Prompt: c.PromptTokens, Completion: c.CompletionTokens,
		Total: c.TotalTokens, CacheRead: c.PromptCacheHit,
	}
	if u.CacheRead == 0 && c.PromptTokensDetails != nil {
		u.CacheRead = c.PromptTokensDetails.CachedTokens
	}
	if c.CompletionTokensDetails != nil {
		u.Reasoning = c.CompletionTokensDetails.ReasoningTokens
	}
	if u.Total == 0 {
		u.Total = u.Prompt + u.Completion
	}
	return u
}

// ParseChat parses a non-streaming chat completion body.
func ParseChat(body []byte) (Usage, bool) {
	var env chatEnvelope
	if err := json.Unmarshal(body, &env); err != nil || env.Usage == nil {
		return Usage{}, false
	}
	if env.Usage.PromptTokens == 0 && env.Usage.CompletionTokens == 0 {
		return Usage{}, false
	}
	return env.Usage.normalize(), true
}

// ParseResponses parses a non-streaming Responses API body.
func ParseResponses(body []byte) (Usage, bool) {
	var env responsesEnvelope
	if err := json.Unmarshal(body, &env); err != nil || env.Response == nil || env.Response.Usage == nil {
		return Usage{}, false
	}
	r := env.Response.Usage
	u := Usage{Prompt: r.InputTokens, Completion: r.OutputTokens, Total: r.TotalTokens}
	if r.InputTokensDetails != nil {
		u.CacheRead = r.InputTokensDetails.CachedTokens
	}
	if r.OutputTokensDetails != nil {
		u.Reasoning = r.OutputTokensDetails.ReasoningTokens
	}
	if u.Total == 0 {
		u.Total = u.Prompt + u.Completion
	}
	return u, true
}

// FromSSEEvent parses one SSE data payload for the given request path.
// isTerminal reports that the stream reached its final event.
func FromSSEEvent(path string, data []byte) (u Usage, found bool, terminal bool) {
	if len(data) == 0 {
		return Usage{}, false, false
	}
	if path == "/v1/responses" {
		var env responsesEnvelope
		if json.Unmarshal(data, &env) != nil {
			return Usage{}, false, false
		}
		if env.Type != "response.completed" || env.Response == nil || env.Response.Usage == nil {
			return Usage{}, false, false
		}
		r := env.Response.Usage
		u = Usage{Prompt: r.InputTokens, Completion: r.OutputTokens, Total: r.TotalTokens}
		if r.InputTokensDetails != nil {
			u.CacheRead = r.InputTokensDetails.CachedTokens
		}
		if r.OutputTokensDetails != nil {
			u.Reasoning = r.OutputTokensDetails.ReasoningTokens
		}
		if u.Total == 0 {
			u.Total = u.Prompt + u.Completion
		}
		return u, true, true
	}
	var env chatEnvelope
	if json.Unmarshal(data, &env) != nil || env.Usage == nil {
		return Usage{}, false, false
	}
	if env.Usage.PromptTokens == 0 && env.Usage.CompletionTokens == 0 {
		return Usage{}, false, false
	}
	return env.Usage.normalize(), true, false
}

const schema = `
CREATE TABLE IF NOT EXISTS requests (
  id INTEGER PRIMARY KEY,
  started_at TEXT NOT NULL,
  completed_at TEXT NOT NULL,
  local_day TEXT NOT NULL,
  path TEXT NOT NULL,
  model TEXT NOT NULL DEFAULT '',
  upstream_model TEXT NOT NULL DEFAULT '',
  stream INTEGER NOT NULL,
  status INTEGER NOT NULL,
  latency_ms INTEGER NOT NULL,
  prompt_tokens INTEGER NOT NULL DEFAULT 0,
  completion_tokens INTEGER NOT NULL DEFAULT 0,

	CostNanoUSD   int64       `json:"cost_nano_usd"`

}

	CompletionTokens int64 `json:"completion_tokens"`
	CacheReadTokens  int64 `json:"cache_read_tokens"`
	CostNanoUSD      int64 `json:"cost_nano_usd"`

}

	Day          string `json:"day"`

	Unpriced         int64  `json:"unpriced"`

type Store struct {
	db    *sql.DB
	ch    chan Record
	flush chan chan struct{}
	wg    sync.WaitGroup
}

func Open(path string) (*Store, error) {
	dsn := fmt.Sprintf("file:%s?_pragma=busy_timeout(5000)&_pragma=journal_mode(WAL)", path)
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, err
	}
	db.SetMaxOpenConns(4)
	if _, err := db.Exec(schema); err != nil {
		db.Close()
		return nil, err
	}
	s := &Store{db: db, ch: make(chan Record, 1024), flush: make(chan chan struct{})}
	s.wg.Add(1)
	go s.writer()
	return s, nil
}

// Enqueue blocks when the queue is full; blocking is preferable to dropping a
// record or writing concurrently with the batch writer.
func (s *Store) Enqueue(r Record) { s.ch <- r }

// Flush blocks until every record enqueued so far is committed.
func (s *Store) Flush() error {
	ack := make(chan struct{})
	s.flush <- ack
	<-ack
	return nil
}

func (s *Store) writer() {
	defer s.wg.Done()
	batch := make([]Record, 0, 200)
	ticker := time.NewTicker(250 * time.Millisecond)
	defer ticker.Stop()
	for {
		select {
		case r, ok := <-s.ch:
			if !ok {
				s.writeBatch(batch)
				return
			}
			batch = append(batch, r)
			if len(batch) >= 200 {
				s.writeBatch(batch)
				batch = batch[:0]
			}
		case <-ticker.C:
			if len(batch) == 0 {
				continue
			}
			s.writeBatch(batch)
			batch = batch[:0]
		case ack := <-s.flush:
			s.writeBatch(batch)
			batch = batch[:0]
			close(ack)
		}
	}
}

func (s *Store) writeBatch(batch []Record) {
	if len(batch) == 0 {
		return
	}
	tx, err := s.db.Begin()
	if err != nil {
		return
	}
	for _, r := range batch {
		if err := insertTx(tx, r); err != nil {
			tx.Rollback()
			return
		}
	}
	_ = tx.Commit()
}

const insertSQL = `INSERT INTO requests
(started_at, completed_at, local_day, path, model, upstream_model, stream, status, latency_ms,
 prompt_tokens, completion_tokens, cache_read_tokens, cache_write_tokens, reasoning_tokens,
 total_tokens, window_label, rates_json, cost_nano_usd, priced, usage_status, error)
VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)`

func args(r Record) []any {
	return []any{
		r.StartedAt.UTC().Format(time.RFC3339Nano),
		r.CompletedAt.UTC().Format(time.RFC3339Nano),
		r.LocalDay, r.Path, r.Model, r.UpstreamModel, boolInt(r.Stream), r.Status, r.LatencyMS,
		r.Usage.Prompt, r.Usage.Completion, r.Usage.CacheRead, r.Usage.CacheWrite,
		r.Usage.Reasoning, r.Usage.Total, r.WindowLabel, r.RatesJSON, r.CostNanoUSD,
		boolInt(r.Priced), r.UsageStatus, r.Error,
	}
}

func insertTx(tx *sql.Tx, r Record) error {
	_, err := tx.Exec(insertSQL, args(r)...)
	return err
}

func boolInt(b bool) int {
	if b {
		return 1
	}
	return 0
}

func (s *Store) Close() error {
	close(s.ch)
	s.wg.Wait()
	if _, err := s.db.Exec("PRAGMA wal_checkpoint(TRUNCATE)"); err != nil {
		s.db.Close()
		return err
	}
	return s.db.Close()
}

func (s *Store) Summary(from, to string) (Summary, error) {
	var out Summary
	err := s.db.QueryRow(`SELECT COUNT(*),
	  COALESCE(SUM(prompt_tokens),0), COALESCE(SUM(completion_tokens),0),
	  COALESCE(SUM(cache_read_tokens),0), COALESCE(SUM(cost_nano_usd),0),
	  COALESCE(SUM(CASE WHEN usage_status != 'ok' THEN 1 ELSE 0 END),0)
	  FROM requests WHERE local_day BETWEEN ? AND ?`, from, to).
		Scan(&out.Requests, &out.PromptTokens, &out.CompletionTokens,
			&out.CacheReadTokens, &out.CostNanoUSD, &out.Unmetered)
	return out, err
}

func (s *Store) Daily(from, to string) ([]Daily, error) {
	rows, err := s.db.Query(`SELECT local_day, COUNT(*),
	  COALESCE(SUM(prompt_tokens),0), COALESCE(SUM(completion_tokens),0),
	  COALESCE(SUM(cost_nano_usd),0)
	  FROM requests WHERE local_day BETWEEN ? AND ?
	  GROUP BY local_day ORDER BY local_day`, from, to)

		return nil, err

// internal/proxy/sse.go
package proxy

import "github.com/kiry163/llmmeter/internal/usage"

const maxLine = 1 << 20

type SSEScanner struct {
	path    string
	onUsage func(usage.Usage)
	buf     []byte
}

func NewSSEScanner(path string, onUsage func(usage.Usage)) *SSEScanner {
	return &SSEScanner{path: path, onUsage: onUsage}
}

func (s *SSEScanner) Write(p []byte) {
	s.buf = append(s.buf, p...)
	for {
		i := indexByte(s.buf, '\n')
		if i < 0 {
			if len(s.buf) > maxLine {
				s.buf = s.buf[:0]
			}
			return
		}
		line := s.buf[:i]
		s.buf = s.buf[i+1:]
		s.handleLine(line)
	}
}

func (s *SSEScanner) handleLine(line []byte) {
	for len(line) > 0 && (line[len(line)-1] == '\r' || line[len(line)-1] == ' ') {
		line = line[:len(line)-1]
	}
	const prefix = "data:"
	if len(line) < len(prefix) || string(line[:len(prefix)]) != prefix {
		return
	}
	payload := line[len(prefix):]
	for len(payload) > 0 && payload[0] == ' ' {
		payload = payload[1:]
	}
	if len(payload) == 0 || string(payload) == "[DONE]" {
		return
	}
	if u, ok, _ := usage.FromSSEEvent(s.path, payload); ok {
		s.onUsage(u)
	}
}

func indexByte(b []byte, c byte) int {
	for i := range b {
		if b[i] == c {
			return i
		}
	}
	return -1
}

// internal/proxy/inject.go
package proxy

import "encoding/json"

// InjectUsage adds stream_options.include_usage for streaming chat requests.
// It returns the original bytes when the body is not JSON, is not streaming,
// or already carries stream_options.
func InjectUsage(body []byte) ([]byte, bool) {
	var req map[string]any
	if err := json.Unmarshal(body, &req); err != nil || req == nil {
		return body, false
	}
	stream, _ := req["stream"].(bool)
	if !stream {
		return body, false
	}
	if _, exists := req["stream_options"]; exists {
		return body, false
	}
	req["stream_options"] = map[string]any{"include_usage": true}
	out, err := json.Marshal(req)
	if err != nil {
		return body, false
	}
	return out, true
}

	var v struct {
		Model string `json:"model"`

	_ = json.Unmarshal(body, &v)

}

	var v struct {
		Model    string `json:"model"`
		Response *struct {
			Model string `json:"model"`
		} `json:"response"`
	}
	_ = json.Unmarshal(body, &v)
	if v.Response != nil && v.Response.Model != "" {
		return v.Response.Model
	}
	return v.Model

// internal/proxy/proxy.go
package proxy

import (
	"bytes"
	"context"
	"crypto/subtle"
	"encoding/json"
	"io"
	"net/http"
	"net/url"
	"strings"
	"sync/atomic"
	"time"

	"github.com/kiry163/llmmeter/internal/config"
	"github.com/kiry163/llmmeter/internal/pricing"
	"github.com/kiry163/llmmeter/internal/store"
	"github.com/kiry163/llmmeter/internal/usage"
)

const (
	maxParsedBody   = 1 << 20
	maxResponseBody = 10 << 20
	drainTimeout    = 3 * time.Second
	drainBytes      = 2 << 20
)

type Proxy struct {
	cfg     atomic.Pointer[config.Config]
	catalog atomic.Pointer[pricing.Catalog]
	loc     atomic.Pointer[time.Location]
	store   *store.Store
	client  *http.Client
	now     func() time.Time
}

func New(cfg *config.Config, cat *pricing.Catalog, st *store.Store, now func() time.Time) (*Proxy, error) {
	if now == nil {
		now = time.Now
	}
	p := &Proxy{store: st, now: now}
	if err := p.Reload(cfg, cat); err != nil {
		return nil, err
	}
	p.client = &http.Client{Transport: &http.Transport{
		Proxy:                 http.ProxyFromEnvironment,
		MaxIdleConnsPerHost:   64,
		IdleConnTimeout:       90 * time.Second,
		ResponseHeaderTimeout: 60 * time.Second,
		ForceAttemptHTTP2:     true,
	}}
	return p, nil
}

func (p *Proxy) Reload(cfg *config.Config, cat *pricing.Catalog) error {
	loc, err := time.LoadLocation(cfg.Timezone)
	if err != nil {
		return err
	}
	p.cfg.Store(cfg)
	p.catalog.Store(cat)
	p.loc.Store(loc)
	return nil
}

// FlushForTest blocks until the store has drained its queue.
func (p *Proxy) FlushForTest() error { return p.store.Flush() }

func (p *Proxy) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	cfg := p.cfg.Load()
	if !validKey(r.Header.Get("Authorization"), cfg.GatewayKey) {
		http.Error(w, `{"error":"unauthorized"}`, http.StatusUnauthorized)
		return
	}
	if !isParsedPath(r.URL.Path) {
		p.forwardRaw(w, r, cfg)
		return
	}
	p.forwardParsed(w, r, cfg)
}

func validKey(h, key string) bool {
	token := strings.TrimSpace(strings.TrimPrefix(h, "Bearer "))
	return subtle.ConstantTimeCompare([]byte(token), []byte(key)) == 1
}

func isParsedPath(path string) bool {
	return path == "/v1/chat/completions" || path == "/v1/responses"
}

func (p *Proxy) forwardRaw(w http.ResponseWriter, r *http.Request, cfg *config.Config) {
	out, cancel, err := p.upstreamRequest(r, cfg, r.Body, r.ContentLength)
	if err != nil {
		http.Error(w, `{"error":"bad upstream"}`, http.StatusBadGateway)
		return
	}
	defer cancel()
	resp, err := p.client.Do(out)
	if err != nil {
		http.Error(w, `{"error":"upstream unreachable"}`, http.StatusBadGateway)
		return
	}
	defer resp.Body.Close()
	copyHeader(w.Header(), resp.Header)
	w.WriteHeader(resp.StatusCode)
	io.Copy(w, resp.Body)
}

// upstreamRequest builds the upstream request. body and length are explicit so
// callers can pass rewritten bytes, a MultiReader, or -1 for chunked encoding.
func (p *Proxy) upstreamRequest(r *http.Request, cfg *config.Config, body io.Reader, length int64) (*http.Request, context.CancelFunc, error) {
	base, err := url.Parse(cfg.Upstream.BaseURL)
	if err != nil {
		return nil, nil, err
	}
	target := *base
	target.Path = strings.TrimRight(base.Path, "/") + r.URL.Path
	target.RawQuery = r.URL.RawQuery
	// WithoutCancel keeps the upstream call alive after the client disconnects,
	// so the usage that the provider already billed can still be read.
	ctx, cancel := context.WithTimeout(context.WithoutCancel(r.Context()), cfg.Upstream.Timeout.Std())
	out, err := http.NewRequestWithContext(ctx, r.Method, target.String(), body)
	if err != nil {
		cancel()
		return nil, nil, err
	}
	copyHeader(out.Header, r.Header)
	out.Header.Del("Accept-Encoding")
	out.Header.Del("Host")
	out.Header.Set("Authorization", "Bearer "+cfg.Upstream.APIKey)
	out.ContentLength = length
	return out, cancel, nil
}

func (p *Proxy) forwardParsed(w http.ResponseWriter, r *http.Request, cfg *config.Config) {
	started := p.now()
	rawBody, err := io.ReadAll(io.LimitReader(r.Body, maxParsedBody+1))
	if err != nil {
		http.Error(w, `{"error":"read body"}`, http.StatusBadRequest)
		return
	}
	parsed := len(rawBody) <= maxParsedBody
	record := store.Record{StartedAt: started, Path: r.URL.Path, UsageStatus: "no_usage"}
	bodyReader := io.Reader(bytes.NewReader(rawBody))
	length := int64(len(rawBody))
	if parsed {
		body := rawBody
		if r.URL.Path == "/v1/chat/completions" {
			if injected, changed := InjectUsage(body); changed {
				body = injected
			}
		}
		bodyReader = bytes.NewReader(body)
		length = int64(len(body))
		record.Model = ParseModel(body)
		record.Stream = isStream(body)
	} else {
		// Preserve the bytes we already consumed: forward them, then the rest.
		record.UsageStatus = "too_large"
		bodyReader = io.MultiReader(bytes.NewReader(rawBody), r.Body)
		length = -1
	}

	out, cancel, err := p.upstreamRequest(r, cfg, bodyReader, length)
	if err != nil {
		http.Error(w, `{"error":"bad upstream"}`, http.StatusBadGateway)
		return
	}
	defer cancel()
	resp, err := p.client.Do(out)
	if err != nil {
		http.Error(w, `{"error":"upstream unreachable"}`, http.StatusBadGateway)
		return
	}
	defer resp.Body.Close()
	record.Status = resp.StatusCode
	copyHeader(w.Header(), resp.Header)
	w.WriteHeader(record.Status)

	var u usage.Usage
	found := false
	contentType := resp.Header.Get("Content-Type")

	switch {
	case strings.Contains(contentType, "text/event-stream"):
		scanner := NewSSEScanner(r.URL.Path, func(v usage.Usage) { u, found = v, true })
		p.streamThrough(w, resp.Body, scanner, &record)
	case strings.Contains(contentType, "json") && parsed:
		buf, tooLarge, bodyErr := readLimited(resp.Body, maxResponseBody)
		if tooLarge {
			record.UsageStatus = "too_large"
			w.Write(buf)
			io.Copy(w, resp.Body)
		} else if bodyErr != nil {
			record.UsageStatus = "parse_failed"
			w.Write(buf)
		} else {
			w.Write(buf)
			if r.URL.Path == "/v1/responses" {
				u, found = usage.ParseResponses(buf)
			} else {
				u, found = usage.ParseChat(buf)
			}
			if m := ParseResponseModel(buf); m != "" {
				record.UpstreamModel = m
			}
		}
	default:
		if _, err := io.Copy(w, resp.Body); err != nil {
			record.UsageStatus = "parse_failed"
		}
	}

	completed := p.now()
	p.finish(&record, cfg, u, found, started, completed)
}

func isStream(body []byte) bool {
	var v struct {
		Stream bool `json:"stream"`
	}
	_ = json.Unmarshal(body, &v)
	return v.Stream
}

func readLimited(r io.Reader, limit int64) ([]byte, bool, error) {
	buf, err := io.ReadAll(io.LimitReader(r, limit+1))
	if int64(len(buf)) > limit {
		return buf, true, nil
	}
	return buf, false, err
}

func (p *Proxy) streamThrough(w http.ResponseWriter, body io.ReadCloser, scanner *SSEScanner, rec *store.Record) {
	flusher, _ := w.(http.Flusher)
	buf := make([]byte, 32*1024)
	clientGone := false
	drained := int64(0)
	var drainTimer *time.Timer
	for {
		n, err := body.Read(buf)
		if n > 0 {
			scanner.Write(buf[:n])
			if !clientGone {
				if _, werr := w.Write(buf[:n]); werr != nil {
					clientGone = true
					// The provider already billed this call; keep reading for a
					// bounded window so the usage can still be recorded.
					drainTimer = time.AfterFunc(drainTimeout, func() { body.Close() })
				} else if flusher != nil {
					flusher.Flush()
				}
			} else {
				drained += int64(n)
				if drained > drainBytes {
					if drainTimer != nil {
						drainTimer.Stop()
					}
					return
				}
			}
		}
		if err != nil {
			if drainTimer != nil {
				drainTimer.Stop()
			}
			return
		}
	}
}

func (p *Proxy) finish(rec *store.Record, cfg *config.Config, u usage.Usage, found bool, started, completed time.Time) {
	loc := p.loc.Load()
	rec.CompletedAt = completed
	rec.LocalDay = completed.In(loc).Format("2006-01-02")
	rec.LatencyMS = completed.Sub(started).Milliseconds()
	if found {
		rec.Usage = u
		rec.UsageStatus = "ok"
		cat := p.catalog.Load()
		lookup, ok := cat.FromYAMLName(rec.Model)
		if !ok {
			rec.Priced = false
			rec.WindowLabel = ""
		} else {
			rates, label, priced := cat.Price(lookup, completed, loc)
			rec.Priced = priced
			rec.WindowLabel = label
			rec.RatesJSON = store.MarshalRates(rates)
			rec.CostNanoUSD = pricing.Cost(rates, u.Prompt, u.Completion, u.CacheRead, u.CacheWrite, u.Reasoning)
		}
	} else if rec.UsageStatus == "" {
		rec.UsageStatus = "no_usage"
	}
	if rec.Status != 0 && rec.Status >= 400 && rec.UsageStatus == "" {
		rec.UsageStatus = "no_usage"
	}
	p.store.Enqueue(*rec)
}

func copyHeader(dst, src http.Header) {
	for k, vv := range src {
		for _, v := range vv {
			dst.Add(k, v)
		}
	}

func New(st *store.Store, loc *time.Location) http.Handler {
	mux := http.NewServeMux()
	mux.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
		if r.URL.Path != "/" {
			http.NotFound(w, r)
			return
		}
		w.Header().Set("Content-Type", "text/html; charset=utf-8")
		w.Write(pageHTML)
	})

		from, to := rangeParams(r, loc)
		sum, err := st.Summary(from, to)
		writeJSON(w, sum, err)
	})
	mux.HandleFunc("/api/daily", func(w http.ResponseWriter, r *http.Request) {
		from, to := rangeParams(r, loc)
		rows, err := st.Daily(from, to)
		writeJSON(w, rows, err)
	})

		from, to := rangeParams(r, loc)
		rows, err := st.ByModel(from, to)
		writeJSON(w, rows, err)
	})
	mux.HandleFunc("/api/requests", func(w http.ResponseWriter, r *http.Request) {
		from, to := rangeParams(r, loc)
		limit, _ := strconv.Atoi(r.URL.Query().Get("limit"))
		offset, _ := strconv.Atoi(r.URL.Query().Get("offset"))
		if limit <= 0 || limit > 200 {
			limit = 50
		}
		rows, err := st.Requests(from, to, r.URL.Query().Get("model"), limit, offset)
		writeJSON(w, rows, err)
	})
	mux.HandleFunc("/api/export.csv", func(w http.ResponseWriter, r *http.Request) {
		from, to := rangeParams(r, loc)
		rows, err := st.AllForExport(from, to, r.URL.Query().Get("model"))
		if err != nil {
			http.Error(w, err.Error(), http.StatusInternalServerError)
			return
		}

		w.Header().Set("Content-Disposition", `attachment; filename="llmmeter.csv"`)
		cw := csv.NewWriter(w)
		cw.Write([]string{"started", "day", "path", "model", "status", "prompt", "completion",
			"cache_read", "reasoning", "window", "cost_usd", "usage_status"})
		for _, it := range rows {
			cw.Write([]string{

				strconv.FormatInt(it.Usage.Reasoning, 10), it.WindowLabel,

			})
		}
		cw.Flush()
	})
	return mux
}

func rangeParams(r *http.Request, loc *time.Location) (string, string) {
	now := time.Now().In(loc)
	to := r.URL.Query().Get("to")
	if to == "" {
		to = now.Format("2006-01-02")
	}
	from := r.URL.Query().Get("from")
	if from == "" {
		from = now.AddDate(0, 0, -6).Format("2006-01-02")
	}
	return from, to
}

func writeJSON(w http.ResponseWriter, v any, err error) {
	w.Header().Set("Content-Type", "application/json")
	if err != nil {
		w.WriteHeader(http.StatusInternalServerError)
		json.NewEncoder(w).Encode(map[string]string{"error": err.Error()})
		return
	}
	json.NewEncoder(w).Encode(v)
}

import (

var version = "dev"

func main() {
	if len(os.Args) > 1 && os.Args[1] == "version" {
		fmt.Println(version)
		return
	}
	if err := serve(os.Args[1:]); err != nil {
		slog.Error("fatal", "err", err)
		os.Exit(1)
	}

func serve(args []string) error {
	fs := flag.NewFlagSet("serve", flag.ExitOnError)
	cfgPath := fs.String("config", "config.yaml", "path to config.yaml")
	dbPath := fs.String("db", "llmmeter.db", "path to sqlite database")
	listen := fs.String("listen", "", "override listen address")
	if err := fs.Parse(args); err != nil {
		return err
	}

	cfg, err := config.Load(*cfgPath)
	if err != nil {
		return err
	}
	cat, err := pricing.New(cfg.Pricing)
	if err != nil {
		return err
	}
	st, err := store.Open(*dbPath)
	if err != nil {
		return err
	}
	defer st.Close()

	p, err := proxy.New(cfg, cat, st, nil)
	if err != nil {
		return err
	}
	loc, err := time.LoadLocation(cfg.Timezone)
	if err != nil {
		return err
	}
	mux := http.NewServeMux()
	mux.Handle("/", p)
	mux.Handle("/api/", web.New(st, loc))
	mux.Handle("/healthz", http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Write([]byte("ok"))
	}))

	if *listen != "" {
		cfg.Listen = *listen
	}
	srv := &http.Server{
		Addr:              cfg.Listen,
		Handler:           mux,
		ReadHeaderTimeout: 10 * time.Second,
		IdleTimeout:       120 * time.Second,
	}

	reload := make(chan os.Signal, 1)
	signal.Notify(reload, syscall.SIGHUP)
	go func() {
		for range reload {
			next, err := config.Load(*cfgPath)
			if err != nil {
				slog.Error("reload failed", "err", err)
				continue
			}
			nextCat, err := pricing.New(next.Pricing)
			if err != nil {
				slog.Error("reload pricing failed", "err", err)
				continue
			}
			if next.Listen != cfg.Listen {
				slog.Warn("listen change requires restart", "current", cfg.Listen, "next", next.Listen)
			}
			if err := p.Reload(next, nextCat); err != nil {
				slog.Error("reload apply failed", "err", err)
				continue
			}
			slog.Info("config reloaded")
		}
	}()

	stop := make(chan os.Signal, 1)
	signal.Notify(stop, syscall.SIGINT, syscall.SIGTERM)
	go func() {
		<-stop
		ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
		defer cancel()
		srv.Shutdown(ctx)
	}()

	slog.Info("llmmeter listening", "addr", cfg.Listen, "upstream", cfg.Upstream.BaseURL)
	if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
		return err
	}
	_ = st.Flush()
	return nil
}
